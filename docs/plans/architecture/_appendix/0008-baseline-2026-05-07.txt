Symptom-confirmation table (decisions per trader, last-hour
window pre-restart)
-----------------------------------------------------------
SELECT t.name, t.latency_class,
       count(d.id) FILTER (WHERE d.created_at > now() - interval '15 minutes') AS d_15m,
       count(d.id) FILTER (WHERE d.created_at > now() - interval '30 minutes') AS d_30m,
       count(d.id) FILTER (WHERE d.created_at > now() - interval '60 minutes') AS d_60m,
       max(d.created_at) AS last_decision
FROM traders t
LEFT JOIN trader_decisions d ON d.trader_id = t.id
WHERE t.is_enabled AND NOT t.is_paused
GROUP BY t.id, t.name, t.latency_class
